\begin{question}{3 (20分)}
    依次解答如下: 
    \begin{itemize}
        \item 二阶无知定义为 $\I\I p := \I(\I p) = \lnot \K(\I p) \land \lnot \K(\lnot \I p)$. 考虑下面这个点模型 $\mathcal{M}, w_0$:
        % https://q.uiver.app/#q=WzAsMyxbMSwwLCJcXGJveGVke3dfMDpwfSJdLFswLDEsIlxcYm94ZWR7d18xOlxcbG5vdCBwfSJdLFsyLDEsIlxcYm94ZWR7d18yOlxcbG5vdCBwfSJdLFswLDBdLFsxLDEsIiIsMix7InJhZGl1cyI6LTN9XSxbMiwyLCIiLDAseyJyYWRpdXMiOi0zfV0sWzAsMV0sWzAsMl1d
        \[\begin{tikzcd}
            & {\boxed{w_0:p}} \\
            {\boxed{w_1:p}} && {\boxed{w_2:\lnot p}}
            \arrow[from=1-2, to=1-2, loop, in=55, out=125, distance=10mm]
            \arrow[from=1-2, to=2-1]
            \arrow[from=1-2, to=2-3]
            \arrow[from=2-1, to=2-1, loop, in=305, out=235, distance=10mm]
            \arrow[from=2-3, to=2-3, loop, in=305, out=235, distance=10mm]
        \end{tikzcd}\]
        其中世界集 $W = \{w_0,w_1, w_2\}$, 可达关系为 $\{(w_0, w_0), (w_0, w_1), (w_0, w_2), (w_1, w_1), (w_2, w_2)\}$.
        赋值为$V(w_0) = \{p\}$, $V(w_1) = \{p\}$, $V(w_2) = \{\lnot p\}$. 可以验证上述可达关系满足自反与传递性.
        \begin{itemize}
            \item 首先证明 $\mathcal{M}, w_0 \models \I p$: 注意到对于 $w_0$ 可达世界为 $\{v|w_0Rv\} = \{w_0, w_1, w_2\}$, 而 $V(w_1) = \{p\}$, $V(w_2) = \{\lnot p\}$, 
            因此有 $\mathcal{M}, w_0 \models \lnot \K p$. 类似的有 $\mathcal{M}, w_0 \models \lnot \K(\lnot p)$. 故 $\mathcal{M}, w_0 \models \I p$.
            \item 又由于$w_1$的可达世界为 $w_1$ 且 $V(w_1) = \{p\}$, 故 $\mathcal{M}, w_1 \models \K p$, 进而有 $\mathcal{M}, w_1 \not\models \I p$.
            类似的我们有 $\mathcal{M}, w_2 \models \K(\lnot p)$, 进而有 $\mathcal{M}, w_2 \not\models \I p$.
        \end{itemize}
        现在我们证明 $\mathcal{M}, w_0 \models \I\I p$: 由于 $\mathcal{M}, w_0 \models \I p$, 且 $\mathcal{M}, w_1 \not\models \I p$, $\mathcal{M}, w_2 \not\models \I p$, 
        即并非所有$w_0$ 的可达世界都满足 $\I p$, 同理并非所有$w_0$ 的可达世界都满足 $\lnot \I p$. 故 
        \begin{align*}
            \mathcal{M}, w_0 \models \lnot\K(\I p) \land \lnot\K(\lnot \I p) \Leftrightarrow \mathcal{M}, w_0 \models \I\I p
        \end{align*}
        \item 对于任意一个点模型 $\M, w \models \I\I p$, 那么有 $\M, w \models \lnot\K\lnot \I p$, 故存在 $u: wRu$ 使得 $\M, u \models \I p$.
        而 $\M, u \models \I p \Leftrightarrow \M, u \models \lnot\K p \land \lnot\K(\lnot p)$, 即
        \begin{gather*}
            \text{存在}~ u_1: uRu_1 \text{ 使得}~ \M, u_1 \models p \\
            \text{存在}~ u_2: uRu_2 \text{ 使得}~ \M, u_2 \models \lnot p
        \end{gather*}
        由传递性知, $wRu_1$且$wRu_2$. 故 
        \begin{align*}
            \M, w \models \lnot\K p \land \lnot\K(\lnot p) \Leftrightarrow \M, w \models \I p
        \end{align*}
        因此, 每一个每个二阶无知 $\I\I p$ 的点模型也都是一阶无知 $\I p$ 的点模型.
        \item 假设存在一个点模型 $\mathcal{M}, w$ 使得 $\mathcal{M}, w \models \K\I\I p$, 那么对于任意 $v: wRv$, 都有 $\mathcal{M}, v \models \I\I p$.
        由于考虑的关系是自反的, 因此有 $wRw$, 那么$\M, w \models \I\I p$, 即
        \begin{align*}
            \M, w \models \lnot\K\I p \land \lnot\K(\lnot \I p) 
        \end{align*}
        那么存在 $u_1, u_2$ 使得 $wRu_1$,$wRu_2$, 且
        \begin{align}
            \label{eq:1}
            \M, u_1 \models \I p, \quad \M, u_2 \models \lnot \I p
        \end{align}
        而由(2)知每个二阶无知 $\I\I p$ 的点模型也都是一阶无知 $\I p$ 的点模型, 故对于任意$v: wRv$都有 $\M, v \models \I p$.
        这与式\eqref{eq:1}矛盾. 故$\K\I\I p$的没有点模型, 即二阶无知不可知.
    \end{itemize}
    \ed
\end{question}
